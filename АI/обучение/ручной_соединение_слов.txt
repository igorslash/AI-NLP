def merge_pair(a, b, splits):
    # Проходим по каждому слову в нашем частотном словаре
    for word in word_freqs:
        split = splits[word]  # Берем текущий список токенов для этого слова
        
        # Если слово состоит всего из одного токена, клеить нечего — пропускаем
        if len(split) == 1:
            continue
        
        i = 0
        # Идем по списку токенов слова (while, потому что длина списка будет меняться)
        while i < len(split) - 1:
            # Если нашли нашу пару токенов (a и b) стоящих рядом...
            if split[i] == a and split[i + 1] == b:
                # 1. Хитрая склейка: 
                # Если второй токен с решетками (##u), берем у него всё кроме первых двух символов.
                # Если без (u), просто склеиваем. Получаем, например, "pu"
                merge = a + b[2:] if b.startswith("##") else a + b
                
                # 2. Обновление списка:
                # Оставляем всё, что было ДО пары + наш новый токен + всё, что было ПОСЛЕ пары
                split = split[:i] + [merge] + split[i + 2 :]
                
                # Мы не увеличиваем i, чтобы на следующем круге проверить 
                # новую склейку с последующими токенами.
            else:
                # Если пара не совпала, просто идем к следующему токену
                i += 1
        
        # Перезаписываем обновленный список токенов для этого слова
        splits[word] = split
        
    return splits
